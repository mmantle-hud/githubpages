# Sorting and Filtering Solutions
```sql
-- 1) Display a unique list of all release years for board games in the database ordered by year of release with the most recent years first. 
SELECT DISTINCT release_year
FROM board_games
ORDER BY release_year DESC;

-- 2) Display the title and year of all the board games that were released in 2016. 
SELECT game_title,
    release_year
FROM board_games
WHERE release_year = 2016;

-- 3) Display the title, year and minimum playing time of all the board games that take at least 45 minutes to play.
SELECT game_title,
    release_year,
    min_playing_time_in_mins
FROM board_games
WHERE min_playing_time_in_mins >= 45;

-- 4) Display the title and numbers of players of all the board games that can be played by 3 players.
SELECT game_title,
    min_num_players,
    max_num_players
FROM board_games
WHERE max_num_players >= 3
    AND min_num_players <= 3;

-- 5)  Display the title, numbers of players and maximum playing time of all the games that can be played in 60 minutes or less by 3 people.
SELECT game_title,
    min_num_players,
    max_num_players,
    max_playing_time_in_mins
FROM board_games
WHERE max_num_players >= 3
    AND min_num_players <= 3
    AND max_playing_time_in_mins <= 60;

-- 6) Display the title, release year and number of players of all the board games that can be played by a single person and were released in 2016. 
SELECT game_title,
    release_year,
    min_num_players,
    max_num_players
FROM board_games
WHERE min_num_players = 1
    AND release_year = 2016;

-- 7) Display the title and year of all the board games that were released in the 1990s. 
SELECT game_title,
    release_year
FROM board_games
WHERE release_year >= 1990
    AND release_year < 2000;

-- 8) Display the title of all the board games that have a title that starts with the word 'The'.
SELECT game_title
FROM board_games
WHERE game_title LIKE 'The %';

-- 9) Display the title and year of all the board games that have a title that starts with the letter 'M' and were released in the 21st century.
SELECT game_title,
    release_year
FROM board_games
WHERE game_title LIKE 'M%'
    AND release_year > 2000;
    
-- 10) Display the title of all the board games that have a title that contains the word 'the'. This should be a case insenstive search.
SELECT game_title
FROM board_games
WHERE game_title ILIKE 'the %'
   OR game_title ILIKE '% the %'
   OR game_title ILIKE '% the';
```
