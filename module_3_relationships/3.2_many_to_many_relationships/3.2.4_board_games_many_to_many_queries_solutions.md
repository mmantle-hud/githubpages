# Many-to-Many Queries Solutions
```sql
-- 1) Display the title and associated game mechanisms for all the board games in the database.
SELECT bg.game_title,
    m.mechanism_name
FROM board_games AS bg
    INNER JOIN board_game_mechanism AS bgm ON bg.board_game_id = bgm.board_game_id
    INNER JOIN mechanisms AS m ON bgm.mechanism_id = m.mechanism_id;
-- 2) Display all the mechanism names for the board game Wingspan
SELECT m.mechanism_name
FROM board_games AS bg
    INNER JOIN board_game_mechanism AS bgm ON bg.board_game_id = bgm.board_game_id
    INNER JOIN mechanisms AS m ON bgm.mechanism_id = m.mechanism_id
WHERE bg.game_title = 'Wingspan';
-- 3) Display the game title and year of release for all the dice rolling games
SELECT bg.game_title,
    bg.release_year
FROM board_games AS bg
    INNER JOIN board_game_mechanism AS bgm ON bg.board_game_id = bgm.board_game_id
    INNER JOIN mechanisms AS m ON bgm.mechanism_id = m.mechanism_id
WHERE m.mechanism_name = 'Dice Rolling';
-- 4) Display the name of each mechanism and the number of games that use each mechanism
SELECT m.mechanism_name,
    COUNT(*) AS num_of_games
FROM mechanisms AS m
    INNER JOIN board_game_mechanism AS bgm ON m.mechanism_id = bgm.mechanism_id
GROUP BY m.mechanism_id,
    m.mechanism_name;
-- 5) Display the title, year of release and number of mechanisms for the board games that have at least two mechanisms
SELECT bg.game_title,
    bg.release_year,
    COUNT(bgm.mechanism_id) AS num_mechanisms
FROM board_games AS bg
    INNER JOIN board_game_mechanism AS bgm ON bg.board_game_id = bgm.board_game_id
GROUP BY bg.board_game_id,
    bg.game_title,
    bg.release_year
HAVING COUNT(bgm.mechanism_id) >= 2;
-- 6) Display the title, year, publisher name and mechanisms for all board games. The results should have one row per board game (use STRING_AGG() to construct a list of mechanisms)
SELECT bg.game_title,
    bg.release_year,
    p.publisher_name,
    STRING_AGG(m.mechanism_name, ', ') AS mechanisms
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
    INNER JOIN board_game_mechanism AS bgm ON bg.board_game_id = bgm.board_game_id
    INNER JOIN mechanisms AS m ON bgm.mechanism_id = m.mechanism_id
GROUP BY bg.board_game_id,
    bg.game_title,
    p.publisher_name;
-- 7) Display the titles and release year of all the games published by Asmodee that use the Set Collection mechanism
SELECT bg.game_title,
    bg.release_year
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
    INNER JOIN board_game_mechanism AS bgm ON bg.board_game_id = bgm.board_game_id
    INNER JOIN mechanisms AS m ON bgm.mechanism_id = m.mechanism_id
WHERE p.publisher_name = 'Asmodee'
    AND m.mechanism_name = 'Set Collection';
-- 8) Display the names of the mechanisms that are used by at least five different board games, along with the number of games that use each mechanism.
SELECT m.mechanism_name,
    COUNT(bgm.board_game_id) AS num_of_games
FROM mechanisms AS m
    INNER JOIN board_game_mechanism AS bgm 
        ON m.mechanism_id = bgm.mechanism_id
GROUP BY m.mechanism_id,
    m.mechanism_name
HAVING COUNT(bgm.board_game_id) >= 5;
```
