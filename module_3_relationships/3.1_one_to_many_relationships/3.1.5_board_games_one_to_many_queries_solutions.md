# One-to-Many Queries Solutions
```sql
-- 1) Display the title, year and publisher name for all the board games released in 2016
SELECT bg.game_title,
    bg.release_year,
    p.publisher_name
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
WHERE bg.release_year = 2016;
-- 2) Display the title, year and publisher name for all the board games released in the 2010s. Order the results by year of release in ascending order.
SELECT bg.game_title,
    bg.release_year,
    p.publisher_name
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
WHERE release_year >= 2010
    AND release_year < 2020
ORDER BY release_year ASC;
-- 3) Display the title and release year for all the games published by Asmodee
SELECT bg.game_title,
    bg.release_year
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
WHERE p.publisher_name = 'Asmodee';
-- 4) Display the title, year, publisher name and publisher country for all the board games where the publisher country isn't 'USA'. Order the results by year of release in ascending order.
SELECT bg.game_title,
    bg.release_year,
    p.publisher_name,
    p.country
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
WHERE p.country <> 'USA'
   OR p.country IS NULL
ORDER BY release_year ASC;
-- 5) Display the title, year and publisher name for all the board games that were not published by Asmodee, Kosmos or Matagot.
SELECT bg.game_title,
    bg.release_year,
    p.publisher_name
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
WHERE p.publisher_name NOT IN ('Asmodee', 'Kosmos', 'Matagot');
-- 6) Display the title and release year of all the board games where the title contains the substring 'Ca' and the publisher is 'Hans im Glück'
SELECT bg.game_title,
    bg.release_year
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
WHERE bg.game_title LIKE '%Ca%'
    AND p.publisher_name = 'Hans im Glück';
-- 7) Display the title, playing time and publisher name of all the board games that can be played by three people. The playing time should be presented as a single string e.g. (60-180)
SELECT bg.game_title,
    CONCAT(
        '(',
        bg.min_playing_time_in_mins,
        '-',
        bg.max_playing_time_in_mins,
        ')'
    ) AS playing_time,
    p.publisher_name
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
WHERE bg.min_num_players <= 3
    AND bg.max_num_players >= 3;
-- 8) Display the title, release year and publisher name of the most recently released board game
SELECT bg.game_title,
    bg.release_year,
    p.publisher_name
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
ORDER BY bg.release_year DESC
LIMIT 1;
-- 9) Display the number of publishers from each country
SELECT country,
    COUNT(*) AS num_publishers
FROM publishers
GROUP BY country;
-- 10) Display the names and number of games of all the publishers that have more than one board game in the database.
SELECT p.publisher_name,
    COUNT(*) AS num_games
FROM board_games AS bg
    INNER JOIN publishers AS p ON bg.publisher_id = p.publisher_id
GROUP BY p.publisher_id,
    p.publisher_name
HAVING COUNT(*) > 1;
```
