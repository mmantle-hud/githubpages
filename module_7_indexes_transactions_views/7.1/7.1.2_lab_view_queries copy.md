-- Run the following query to create the view
CREATE VIEW flight_details AS
SELECT
    f.flight_id,
    f.departure,
    f.arrival,
    f.base_price,
    r.route_id,
    r.distance_km,
    origin.iata_code AS origin_code,
    origin.name AS origin_name,
    origin.city AS origin_city,
    dest.iata_code AS destination_code,
    dest.name AS destination_name,
    dest.city AS destination_city,
    a.tail_number,
    am.manufacturer,
    am.model_name,
    am.seat_capacity
FROM flights AS f
JOIN routes AS r
    ON f.route_id = r.route_id
JOIN airports AS origin
    ON r.origin_airport = origin.iata_code
JOIN airports AS dest
    ON r.destination_airport = dest.iata_code
LEFT JOIN aircraft AS a
    ON f.tail_number = a.tail_number
LEFT JOIN aircraft_models AS am
    ON a.model_id = am.model_id;

-- Using SQLTools check it has worked

-- route_summary — aggregate view showing distance and flight count per route (GROUP BY route_id), good for teaching views combined with aggregation.
SELECT origin_name, destination_name, COUNT(*) FROM flight_details
GROUP BY route_id, origin_name, destination_name;
